#!/usr/bin/env python3
"""Run 1-needle & 4-needle WikiText-103 NIAH + 200-count Benchmark Suite + Graph + HF upload for Bolt-3."""
from __future__ import annotations

import glob
import json
import os
import random
import re
import sys
from pathlib import Path


def load_env_tokens() -> dict[str, str]:
    found = {}
    for p in ("/marimo/.env", ".env", "../.env", "../../.env", os.path.expanduser("~/.env")):
        if not os.path.isfile(p):
            continue
        with open(p, "r", encoding="utf-8", errors="ignore") as f:
            for line in f:
                s = line.strip()
                if not s or s.startswith("#") or "=" not in s:
                    continue
                k, v = s.split("=", 1)
                k, v = k.strip(), v.strip().strip('"').strip("'")
                if not v:
                    continue
                if ("GITHUB" in k.upper() or "GH_TOKEN" in k.upper() or v.startswith(("ghp_", "github_pat_"))) and "GITHUB_TOKEN" not in found:
                    found["GITHUB_TOKEN"] = v
                if ("HF" in k.upper() or "HUGGING" in k.upper() or v.startswith("hf_")) and "HF_TOKEN" not in found:
                    found["HF_TOKEN"] = v
    if "HF_TOKEN" in found:
        os.environ["HF_TOKEN"] = found["HF_TOKEN"]
        os.environ["HUGGING_FACE_HUB_TOKEN"] = found["HF_TOKEN"]
    return found


def find_repo_and_ckpt() -> tuple[str, str]:
    local_root = Path(__file__).resolve().parent.parent if "__file__" in globals() else Path.cwd()
    if not (local_root / "bolt3").is_dir() and (local_root.parent / "bolt3").is_dir():
        local_root = local_root.parent

    for pat in ("bolt3_b_qwen3_4b_latest_benchmarks.json", "bolt3_b_qwen3_4b_latest.pth", "bolt3_qwen3_4b_latest.pth"):
        hits = sorted(glob.glob(f"/marimo/**/{pat}", recursive=True), key=len, reverse=True)
        if hits:
            wd = os.path.dirname(hits[0])
            ckpt = os.path.join(wd, "bolt3_b_qwen3_4b_latest.pth")
            if not os.path.isfile(ckpt):
                ckpt = os.path.join(wd, "bolt3_qwen3_4b_latest.pth")
            repo = wd if os.path.isfile(os.path.join(wd, "bolt3/benchmark.py")) else str(local_root)
            return repo, ckpt

    repo_hits = sorted(glob.glob("/marimo/**/bolt3/benchmark.py", recursive=True))
    repo = os.path.dirname(os.path.dirname(repo_hits[0])) if repo_hits else str(local_root)
    assert os.path.isfile(os.path.join(repo, "bolt3/benchmark.py")), "Cannot locate bolt3/benchmark.py"
    ckpt = os.path.join(repo, "checkpoints", "bolt3_b_qwen3_4b_latest.pth")
    if not os.path.isfile(ckpt):
        ckpt = os.path.join(repo, "bolt3_b_qwen3_4b_latest.pth")
    return repo, ckpt


def main():
    tokens = load_env_tokens()
    repo, healed_ckpt = find_repo_and_ckpt()
    work_dir = os.path.dirname(healed_ckpt) or repo
    os.makedirs(work_dir, exist_ok=True)
    os.chdir(repo)

    sys.path[:] = [repo] + [p for p in sys.path if p != repo]
    for mod_k in list(sys.modules.keys()):
        if mod_k == "bolt3" or mod_k.startswith("bolt3."):
            del sys.modules[mod_k]

    import torch
    from datasets import load_dataset
    from transformers import AutoTokenizer
    from bolt3.attention import Bolt3Decoder
    from bolt3.benchmark import run_benchmark_suite
    from bolt3.converter import _purge_zombie_vram
    from bolt3.model import load_bolt3_checkpoint

    if not os.path.isfile(healed_ckpt):
        from huggingface_hub import hf_hub_download

        healed_ckpt = hf_hub_download("rikkathree/bolt-qwen3-4b", filename="bolt3_b_qwen3_4b_latest.pth", local_dir=work_dir)

    niah_cache = os.path.join(work_dir, "bolt3_b_niah_cache.json")
    if os.path.isfile(niah_cache):
        with open(niah_cache) as f:
            nc = json.load(f)
        niah_1_scores = {int(k): v for k, v in nc["niah_1"].items()}
        niah_4_scores = {int(k): v for k, v in nc["niah_4"].items()}
    else:
        tok = AutoTokenizer.from_pretrained("Qwen/Qwen3-4B")
        wiki_ids = []
        for r in load_dataset("Salesforce/wikitext", "wikitext-103-raw-v1", split="test", streaming=True):
            if r["text"].strip():
                wiki_ids.extend(tok.encode(r["text"], add_special_tokens=False))
                if len(wiki_ids) >= 36000:
                    break
        model, _, _, _, _ = load_bolt3_checkpoint(healed_ckpt, device="cuda", eval_mode=True)
        dec = Bolt3Decoder(model, max_cache_pos=36000)

        # Batched 1-Needle NIAH across all 4 trials per context length (4x faster)
        niah_1_scores = {}
        pre = tok.encode("<|im_start|>user\nRemember the secret passkey in the text:\n\n", add_special_tokens=False)
        query = tok.encode(
            "\nQuestion: What is the secret passkey? Reply with only the 5 digits.<|im_end|>\n<|im_start|>assistant\n<think>\n\n</think>\n\nThe secret passkey is ",
            add_special_tokens=False,
        )
        for ctx_len in (2048, 4096, 8192, 16384):
            batch_rows, codes = [], []
            for t in range(4):
                code = str(random.Random(1000 + ctx_len + t).randint(10000, 99999))
                needle = tok.encode(f"\nImportant memo: The secret passkey is {code}. Remember {code}.\n", add_special_tokens=False)
                n_fill = max(64, ctx_len - len(pre) - len(needle) - len(query))
                hay = list(wiki_ids[:n_fill])
                ins = int(n_fill * 0.15)
                row = (pre + hay[:ins] + needle + hay[ins:] + query)[:ctx_len]
                batch_rows.append(row)
                codes.append(code)
            min_len = min(len(r) for r in batch_rows)
            batch_rows = [r[:min_len] for r in batch_rows]
            toks, _ = dec.generate_batch(torch.tensor(batch_rows, device="cuda"), max_new_tokens=12, temperature=0.0)
            hits = sum(int(codes[i] in tok.decode(toks[i].tolist(), skip_special_tokens=True)) for i in range(4))
            niah_1_scores[ctx_len] = 100.0 * hits / 4.0

        niah_4_scores = {}
        facts = [
            ("Project Orion", "launch code", "48291"),
            ("Protocol Hemlock", "temp", "34715"),
            ("Station Vanguard", "freq", "88450"),
            ("Facility Chronos", "key", "61904"),
        ]
        pre_c = tok.encode("<|im_start|>user\nRead carefully and remember every secret record:\n\n", add_special_tokens=False)
        for L in (4096, 16384, 32768):
            hay = list(pre_c) + list(wiki_ids[: L - len(pre_c)])
            for (proj, prop, val), d in zip(facts, (0.10, 0.35, 0.65, 0.85)):
                nd = tok.encode(f" [RECORD: The {prop} for {proj} is {val}. Remember {val}.] ", add_special_tokens=False)
                pos = min(L - len(nd) - 16, max(len(pre_c) + 16, int(L * d)))
                hay[pos : pos + len(nd)] = nd
            _, st0, kv0 = dec.prefill(torch.tensor([hay], device="cuda"), max_kv_len=L + 256)
            hits = []
            for proj, prop, val in facts:
                stc = [s.clone() if s is not None else None for s in st0]
                kvc = [{k: (v.clone() if torch.is_tensor(v) else v) for k, v in item.items()} for item in kv0]
                qs = f"\n\nQuestion: What is the {prop} for {proj}? Reply with only the 5-digit number.<|im_end|>\n<|im_start|>assistant\n<think>\n\n</think>\n\nThe 5-digit number is "
                qt = torch.tensor([tok.encode(qs, add_special_tokens=False)], device="cuda")
                toks, _, _, _ = dec.continue_generate(qt, stc, kvc, L, 10)
                m = re.search(r"\d+", tok.decode(toks[0].tolist(), skip_special_tokens=True))
                hits.append(int(bool(m) and m.group(0) == val))
            niah_4_scores[L] = 100.0 * sum(hits) / 4.0

        with open(niah_cache, "w") as f:
            json.dump({"niah_1": niah_1_scores, "niah_4": niah_4_scores}, f, indent=2)
        del model, dec
        _purge_zombie_vram()

    results = run_benchmark_suite(healed_ckpt, donor_id="Qwen/Qwen3-4B", compare_donor=True, limit=200, gsm8k_limit=200)

    import matplotlib

    matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    import numpy as np

    plt.style.use("dark_background")
    fig, axes = plt.subplots(2, 2, figsize=(15, 9.5), dpi=170)
    fig.patch.set_facecolor("#0d1117")
    for ax in axes.flat:
        ax.set_facecolor("#161b22")
        ax.grid(True, alpha=0.2, color="#8b949e", linestyle="--")

    tasks = list(results["student"].keys())
    x_pos = np.arange(len(tasks))
    w = 0.36
    axes[0, 0].bar(x_pos - w / 2, [results["teacher"].get(k, {}).get("acc", 0.0) for k in tasks], w, label="Qwen3-4B Donor (100% Softmax)", color="#58a6ff")
    axes[0, 0].bar(x_pos + w / 2, [results["student"][k]["acc"] for k in tasks], w, label="Bolt-3 (36/36 Bolt-3, 0 Softmax)", color="#3fb950")
    axes[0, 0].set_xticks(x_pos)
    axes[0, 0].set_xticklabels(tasks, rotation=15, fontsize=8.5, fontweight="bold")
    axes[0, 0].set_ylim(0, 105)
    axes[0, 0].set_ylabel("Accuracy (%)", fontweight="bold")
    axes[0, 0].set_title("1. N=200 Real-World Suite vs Official Donor", fontweight="bold")
    axes[0, 0].legend(loc="upper right", fontsize=8)

    n_labels = [f"1N@{k//1024}k" for k in niah_1_scores] + [f"4N@{k//1024}k" for k in niah_4_scores]
    n_vals = list(niah_1_scores.values()) + list(niah_4_scores.values())
    bars = axes[0, 1].bar(n_labels, n_vals, color=["#3fb950"] * len(niah_1_scores) + ["#a371f7"] * len(niah_4_scores), width=0.55)
    axes[0, 1].set_ylim(0, 115)
    axes[0, 1].set_ylabel("Exact Recall (%)", fontweight="bold")
    axes[0, 1].set_title("2. Non-Repeating WikiText-103 NIAH (1-Needle & 4-Needle)", fontweight="bold")
    for b, v in zip(bars, n_vals):
        axes[0, 1].text(b.get_x() + b.get_width() / 2, v + 2.5, f"{v:.0f}%", ha="center", fontweight="bold", fontsize=8.5)

    sc = results["systems"]["scaling"]
    ctxs = [f"{r['context']//1024}k" for r in sc]
    stu_ram = [r.get("bolt3_state_mb", r.get("vsa_kv_mb", 0.0)) for r in sc]
    stu_tps = [r.get("bolt3_tok_s", r.get("vsa_tok_s", 0.0)) for r in sc]
    axes[1, 0].plot(ctxs, [r["donor_kv_mb"] for r in sc], "o--", color="#f85149", lw=2.2, label="Donor Softmax KV (Unbounded)")
    axes[1, 0].plot(ctxs, stu_ram, "s-", color="#3fb950", lw=2.5, label="Bolt-3 Fixed State (O(1))")
    axes[1, 0].fill_between(range(len(ctxs)), stu_ram, [r["donor_kv_mb"] for r in sc], color="#3fb950", alpha=0.16)
    axes[1, 0].set_xlabel("Context Length", fontweight="bold")
    axes[1, 0].set_ylabel("State / Cache RAM (MB)", fontweight="bold")
    axes[1, 0].set_title(f"3. Constant State RAM (-{sc[-1]['ram_saved_pct']}% @ 64k)", fontweight="bold")
    axes[1, 0].legend(loc="upper left", fontsize=8)

    axes[1, 1].plot(ctxs, [r["donor_tok_s"] for r in sc], "o--", color="#58a6ff", lw=2.2, label="Donor Softmax")
    axes[1, 1].plot(ctxs, stu_tps, "s-", color="#d29922", lw=2.5, label="Bolt-3 (36/36 Full Bolt-3)")
    axes[1, 1].set_xlabel("Context Length", fontweight="bold")
    axes[1, 1].set_ylabel("Decode Speed (tok/s)", fontweight="bold")
    axes[1, 1].set_title(f"4. Decode Throughput ({sc[-1]['speedup']}x @ 64k)", fontweight="bold")
    axes[1, 1].legend(loc="best", fontsize=8)

    plt.suptitle("Bolt-3 (36/36 Full Bolt, 0 Softmax) vs Qwen/Qwen3-4B — Full Evaluation", fontsize=13, fontweight="bold", y=0.98)
    plt.tight_layout(rect=[0, 0, 1, 0.95])
    out_chart = os.path.join(work_dir, "bolt3_infinite_report.png")
    plt.savefig(out_chart, dpi=170)
    plt.close(fig)
    print(f"[chart] Saved complete summary graph -> {out_chart}", flush=True)

    if "HF_TOKEN" in tokens:
        from huggingface_hub import HfApi

        api = HfApi(token=tokens["HF_TOKEN"])
        base = os.path.splitext(healed_ckpt)[0]
        for fp in (
            healed_ckpt,
            base + "_benchmarks.json",
            base + "_benchmark.png",
            out_chart,
        ):
            if os.path.isfile(fp):
                api.upload_file(path_or_fileobj=fp, path_in_repo=os.path.basename(fp), repo_id="rikkathree/bolt-qwen3-4b", repo_type="model")
        print("[hf] Uploaded checkpoint + benchmarks + graphs -> https://huggingface.co/rikkathree/bolt-qwen3-4b", flush=True)


if __name__ == "__main__":
    main()
